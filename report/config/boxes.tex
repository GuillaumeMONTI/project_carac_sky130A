% ============================================================
% COMMON TECHNICAL BOX STYLE
% ============================================================

\tcbset{
    technicalbox/.style={
        enhanced,
        breakable,
        boxrule=0pt,
        arc=1mm,
        left=4mm,
        right=4mm,
        top=3mm,
        bottom=3mm,
        before skip=1em,
        after skip=1em,
        fonttitle=\sffamily\bfseries,
        coltitle=TextColor
    }
}
